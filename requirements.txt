# Core framework
fastapi
uvicorn[standard]          # you need the server in production unless you deploy under gunicorn or uvicorn workers
pydantic                   # runtime dependency of FastAPI and Strawberry
pydantic-settings           # required if you use it for environment/config loading
python-dotenv              # if your settings rely on .env files in production (common)

# HTTP client / DI
httpx
dependency-injector        # used at runtime for DI container
httpx[http2]

# Database and migrations
sqlalchemy[asyncio]
asyncpg                    # PostgreSQL driver
alembic                    # needed in prod *only* if you run migrations there (most do)

# Cache
redis[hiredis]

# Encode / crypto
bcrypt

# GraphQL
strawberry-graphql[fastapi]

# Security & login
python-jose
google-auth
google-auth-oauthlib
requests                   # runtime for Google OAuth and other HTTP calls
